\newglossaryentry{llm}{
  name={LLM},
  description={large language model: a neural network trained on a large
    collection of text to understand and generate free-form language}
}

\newglossaryentry{semanticentropy}{
  name={semantic entropy},
  description={an uncertainty score for free-form answers. The model answers
    the same prompt several times, answers that mean the same thing are grouped
    using entailment tests, and Shannon entropy measures how widely the samples
    are spread across the groups. A low value means that the samples mostly
    express one meaning; a high value means that they express conflicting
    meanings. The score is computed after training and does not change the
    model's weights}
}

\newglossaryentry{surfaceentropy}{
  name={surface entropy},
  description={an uncertainty score that groups sampled answers only when their
    normalized text matches exactly. It measures variation in wording rather
    than variation in meaning. The same calculation without normalization is
    called \emph{naive sample entropy}}
}

\newglossaryentry{ptrue}{
  name={P(True)},
  description={a self-evaluation method that asks the model whether its answer
    is correct, using sampled alternative answers as context. The uncertainty
    score is one minus the probability of the ``true'' option, without
    renormalizing over the two options, following the released code}
}

\newglossaryentry{confabulation}{
  name={confabulation},
  description={a wrong answer that changes in meaning when the same prompt is
    sampled repeatedly. This differs from a systematic error, where the model
    gives the same wrong answer each time}
}

\newglossaryentry{hallucination}{
  name={hallucination},
  description={a broad term for an output that is fluent but unsupported,
    factually wrong, or unfaithful to a source. This thesis focuses on
    confabulations that can be detected by comparing repeated samples}
}

\newglossaryentry{nli}{
  name={NLI},
  description={Natural Language Inference: a task in which a model compares a
    premise with a hypothesis and labels the hypothesis as implied,
    contradicted, or neither. This thesis uses NLI to judge whether two sampled
    answers mean the same thing}
}

\newglossaryentry{entailment}{
  name={entailment},
  description={an NLI label meaning that one sentence logically implies
    another. This thesis treats two answers as equivalent when each entails the
    other}
}


\newglossaryentry{auroc}{
  name={AUROC},
  description={area under the receiver operating characteristic curve: a
    measure of how well a score ranks wrong answers above correct ones across
    all possible decision thresholds. A higher value means better ranking}
}

\newglossaryentry{aurac}{
  name={AURAC},
  description={area under the rejection-accuracy curve: a measure of how
    accuracy changes as the answers with the highest uncertainty are removed
    one by one. Higher is better, but values can be compared only for the same
    task and model}
}


\newglossaryentry{deberta}{
  name={DeBERTa},
  description={Decoding-enhanced BERT with disentangled attention: a
    Transformer model designed to understand relationships between texts. This
    thesis uses DeBERTa models to group answers through entailment}
}

\newglossaryentry{sep}{
  name={SEP},
  description={Semantic Entropy Probe: a small linear classifier that reads a
    language model's fixed internal states and predicts high semantic entropy.
    It avoids generating several answers each time the score is needed}
}
