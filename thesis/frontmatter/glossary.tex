\newglossaryentry{llm}{
  name={LLM},
  description={large language model: a neural language model trained on large
    text corpora and capable of generating fluent free-form text in response
    to prompts}
}

\newglossaryentry{semanticentropy}{
  name={semantic entropy},
  description={an uncertainty score for free-form outputs: the Shannon entropy
    over the empirical distribution of semantic clusters, obtained by grouping
    repeated sampled answers under bidirectional NLI entailment. Low values
    mean the samples express one meaning, high values that they spread across
    incompatible ones. It is computed after training and never changes the
    model's weights}
}

\newglossaryentry{surfaceentropy}{
  name={surface entropy},
  description={the Shannon entropy over sampled answers reduced to their
    normalized strings, so two answers coincide only when those strings are
    identical. With the unnormalized \emph{naive sample entropy} it forms the
    surface-form baseline: it measures how much the wording varies, not what
    the answers mean}
}

\newglossaryentry{ptrue}{
  name={P(True)},
  description={a single-pass self-evaluation baseline: the model is shown its
    own most-likely answer and asked how likely it is to be true. The
    probability it assigns to ``True'' gives the uncertainty score
    $1-P(\mathrm{True})$. Needs no repeated sampling}
}

\newglossaryentry{confabulation}{
  name={confabulation},
  description={a hallucination subtype in which the answer is both incorrect
    and semantically inconsistent across repeated samplings of the same
    prompt. Distinguished from systematic errors, where the model gives the
    same wrong answer confidently every time}
}

\newglossaryentry{hallucination}{
  name={hallucination},
  description={a broad term for outputs that are fluent but unsupported,
    factually incorrect, or unfaithful to a source. Here it follows the
    semantic entropy literature and refers primarily to confabulation-type
    failures detectable by sampling inconsistency}
}

\newglossaryentry{nli}{
  name={NLI},
  description={Natural Language Inference: deciding whether a hypothesis
    sentence is entailed by, contradicted by, or neutral with respect to a
    premise. Used here as the backend for judging whether two sampled answers
    mean the same thing}
}

\newglossaryentry{entailment}{
  name={entailment},
  description={an NLI label indicating that the premise logically implies the
    hypothesis. Bidirectional entailment, where each sentence entails the
    other, is used as the proxy for semantic equivalence}
}


\newglossaryentry{auroc}{
  name={AUROC},
  description={area under the receiver operating characteristic curve: a
    threshold-free measure of ranking quality. Incorrect answers are the
    positive class, so a higher AUROC means the estimator ranks incorrect
    answers above correct ones more reliably}
}

\newglossaryentry{aurac}{
  name={AURAC},
  description={area under the rejection-accuracy curve: a selective-prediction
    metric computed by progressively withholding the highest-uncertainty
    answers and measuring accuracy on the rest. Higher is better, but values
    are comparable only within one task and model}
}


\newglossaryentry{deberta}{
  name={DeBERTa},
  description={Decoding-enhanced BERT with disentangled attention: a
    Transformer model with strong NLI benchmark performance, used here as the
    local backend for entailment-based clustering}
}

\newglossaryentry{sep}{
  name={SEP},
  description={Semantic Entropy Probe: a lightweight linear classifier on the
    generator's frozen hidden states, trained to predict high semantic entropy
    without multi-sample generation at inference time}
}
